-Run in foreground (easy to watch logs)
cd /root/dash-indexer
export DATABASE_URL="sqlite://dashymarket_indexer.db"
export RUST_LOG="dash_indexer=info"
./target/release/dash-indexer start --start-block 2395285 --port 3838

-Run in background (log + pid)
cd /root/dash-indexer
export DATABASE_URL="sqlite://dashymarket_indexer.db"
export RUST_LOG="dash_indexer=info"
nohup ./target/release/dash-indexer start --start-block 2395285 --port 3838 > indexer_dashymarket.log 2>&1 &
echo $! > indexer_dashymarket.pid

-Verify it’s working
curl http://localhost:3838/health
curl http://localhost:3838/status
